cmake_minimum_required(VERSION 3.20)
project(CrusaderEnv LANGUAGES CXX)
add_library(crusader_observation INTERFACE)
target_include_directories(crusader_observation INTERFACE src)
target_compile_features(crusader_observation INTERFACE cxx_std_20)
include(CTest)
if(BUILD_TESTING)
  add_executable(observation_test tests/observation_test.cpp)
  target_link_libraries(observation_test PRIVATE crusader_observation)
  if(MSVC)
    target_compile_options(observation_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(observation_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME observation_decode COMMAND observation_test)
endif()
if(BUILD_TESTING)
  add_executable(render_observation_test tests/render_observation_test.cpp)
  target_link_libraries(render_observation_test PRIVATE crusader_observation)
  if(MSVC)
    target_compile_options(render_observation_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(render_observation_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME render_observation_decode COMMAND render_observation_test)
endif()
if(WIN32)
  add_executable(crusader_probe src/windows/probe.cpp)
  target_compile_features(crusader_probe PRIVATE cxx_std_20)
  target_link_libraries(crusader_probe PRIVATE bcrypt)
  if(MINGW)
    target_link_options(crusader_probe PRIVATE -municode)
  endif()
endif()
if(WIN32)
  add_library(crusader_managed_snapshot OBJECT src/windows/managed_snapshot.cpp)
  target_link_libraries(crusader_managed_snapshot PRIVATE crusader_observation)
endif()
if(WIN32)
  add_library(crusader_adapter SHARED src/windows/adapter.cpp)
  target_compile_features(crusader_adapter PRIVATE cxx_std_20)
  target_link_libraries(crusader_adapter PRIVATE crusader_observation)
  set_target_properties(crusader_adapter PROPERTIES PREFIX "")
  if(MINGW)
    target_link_options(crusader_adapter PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
if(WIN32)
  add_executable(crusader_launcher src/windows/probe.cpp)
  target_compile_features(crusader_launcher PRIVATE cxx_std_20)
  target_compile_definitions(crusader_launcher PRIVATE CRUSADER_LAUNCHER)
  target_link_libraries(crusader_launcher PRIVATE bcrypt)
  add_dependencies(crusader_launcher crusader_adapter)
  if(MINGW)
    target_link_options(crusader_launcher PRIVATE -municode)
  endif()
endif()

if(WIN32 AND BUILD_TESTING)
  add_executable(adapter_contract_test tests/adapter_contract_test.cpp)
  target_link_libraries(adapter_contract_test PRIVATE crusader_adapter crusader_observation)
  if(MSVC)
    target_compile_options(adapter_contract_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(adapter_contract_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME adapter_contract COMMAND adapter_contract_test)
endif()
